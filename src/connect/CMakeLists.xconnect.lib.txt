NCBI_begin_lib(xconnect)
  if(NCBI_DLL_BUILD AND NOT NCBI_PTBCFG_ALLOW_COMPOSITE
     AND MaxDebug IN_LIST NCBI_PTBCFG_PROJECT_FEATURES)
    set(extra_lib connect)
    set(extra_src "")
  else()
    set(extra_lib "")
    set(extra_src ${SRC_C})
  endif()
  NCBI_sources(${extra_src} ${SRC_CXX})
  NCBI_headers(*.hpp)
  NCBI_add_include_directories(${CMAKE_CURRENT_LIST_DIR}/mbedtls)
  NCBI_optional_components(GNUTLS)
  NCBI_uses_toolkit_libraries(${extra_lib} xncbi)
  NCBI_uses_external_libraries(${NETWORK_LIBS})
  NCBI_project_tags(core)
  NCBI_project_watchers(lavr)
NCBI_end_lib()

if(CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM"
   AND NOT CMAKE_BUILD_TYPE STREQUAL "Debug")
    set_source_files_properties(${NCBI_CURRENT_SOURCE_DIR}/mbedtls/bignum.c
        PROPERTIES COMPILE_OPTIONS "-O")
endif()
