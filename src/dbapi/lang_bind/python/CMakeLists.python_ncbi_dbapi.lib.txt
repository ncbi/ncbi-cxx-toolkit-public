NCBI_begin_lib(python_ncbi_dbapi)
  NCBI_sources(python_ncbi_dbapi)
  NCBI_headers(*.hpp pythonpp/*.hpp)
  NCBI_uses_toolkit_libraries(dbapi xutil)
  NCBI_optional_components(connext)
  NCBI_project_watchers(ucko satskyse)
NCBI_end_lib(_result)

if(_result AND UNIX AND BUILD_SHARED_LIBS)
 function(make_pydbapi_link _cfg)
  add_custom_command(TARGET python_ncbi_dbapi POST_BUILD
    COMMAND ln -f -s libpython_ncbi_dbapi${CMAKE_SHARED_LIBRARY_SUFFIX}
                     ./${_cfg}/python_ncbi_dbapi.so
    BYPRODUCTS ${CMAKE_LIBRARY_OUTPUT_DIRECTORY}/${_cfg}/python_ncbi_dbapi.so
    WORKING_DIRECTORY ${CMAKE_LIBRARY_OUTPUT_DIRECTORY}
  )
 endfunction()
 if(NCBI_GENERATOR_IS_MULTI_CONFIG)
  foreach(_cfg IN LISTS NCBI_CONFIGURATION_TYPES)
   make_pydbapi_link(${_cfg})
  endforeach()
 else()
  make_pydbapi_link("")
 endif()
endif()
